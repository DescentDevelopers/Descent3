set(CMAKE_FOLDER "netcon")

include_directories("includes")
add_subdirectory(inetfile)
add_subdirectory(lanclient)
add_subdirectory(mtclient)
add_subdirectory(descent3onlineclient)

if(CMAKE_SYSTEM_NAME STREQUAL "Android")
  # On Android the connection modules aren't wrapped in online/*.d3c files: they're packaged in the
  # APK as native libraries and loaded from there by name. Tell the game which ones there are.
  set(D3_BUNDLED_NETCON_ENTRIES "")
  foreach(target Parallax_Online Descent3_Online_TCP_IP Direct_TCP_IP)
    get_target_property(name ${target} OUTPUT_NAME)
    string(APPEND D3_BUNDLED_NETCON_ENTRIES "\n    \"${name}\",")
  endforeach()
  file(CONFIGURE
    OUTPUT "${PROJECT_BINARY_DIR}/lib/d3_bundled_netcon.h"
    CONTENT "// Generated by netcon/CMakeLists.txt.\n// Connection modules packaged as native libraries rather than as online/*.d3c files (Android).\n#pragma once\n\ninline constexpr const char *D3_BUNDLED_NETCON_MODULES[] = {@D3_BUNDLED_NETCON_ENTRIES@\n};\n"
    @ONLY
  )
endif()
